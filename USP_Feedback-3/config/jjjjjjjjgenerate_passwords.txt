<?php
/**
 * Password Hash Generator
 * Run this file once to generate hashes, then delete it
 */

// Function to generate password hash
function generatePasswordHash($plainPassword) {
    return password_hash($plainPassword, PASSWORD_BCRYPT, ['cost' => 10]);
}

// Generate hashes for demo passwords
$passwords = [
    'admin123' => generatePasswordHash('admin123'),
    'staff123' => generatePasswordHash('staff123'),
];

echo "=== PASSWORD HASHES ===\n\n";

foreach ($passwords as $plain => $hash) {
    echo "Plain Password: $plain\n";
    echo "Hash: $hash\n";
    echo "---\n\n";
}

echo "\n=== SQL UPDATE STATEMENTS ===\n\n";

echo "UPDATE users SET password = '" . $passwords['admin123'] . "' WHERE email = 'admin@usp.com';\n\n";
echo "UPDATE users SET password = '" . $passwords['staff123'] . "' WHERE email = 'staff@usp.com';\n";

?>
